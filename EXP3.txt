package TEETC1234;
import javax.swing.*; 
import java.awt.*; 
import java.awt.event.*; 

public class exp3 { 
   // Function to write student information in JFrame and store it in a file 
   public static void StudentInfo() 
   {
       JFrame f = new JFrame("Student Result Form");   

       // Creating the labels 
       JLabel l1, l2, l3, l4, l5, l6, l7, l8, l9; 

       // Creating text fields for student name, roll no, and marks 
       JTextField t1, t2, t3, t4, t5, t6, t7; 

       // Creating two JComboBoxes for Branch and Division 
       JComboBox<String> j1, j2; 

       // Creating two buttons 
       JButton b1, b2; 

       // Naming the labels and setting bounds
       l1 = new JLabel("Student Name:");
       l1.setBounds(50, 50, 100, 30);
       l2 = new JLabel("Branch:");
       l2.setBounds(50, 100, 120, 30);
       l3 = new JLabel("Div:");
       l3.setBounds(400, 50, 50, 30);
       l4 = new JLabel("Roll no:");
       l4.setBounds(400, 100, 70, 30);
       l5 = new JLabel("MC:");
       l5.setBounds(400, 150, 70, 30);
       l6 = new JLabel("EMF:");
       l6.setBounds(50, 150, 70, 30);
       l7 = new JLabel("FJP:");
       l7.setBounds(400, 200, 70, 30);
       l8 = new JLabel("DC:");
       l8.setBounds(50, 200, 70, 30);
       l9 = new JLabel("DBMS:");
       l9.setBounds(400, 250, 70, 30);

       // TextFields
       t1 = new JTextField();
       t1.setBounds(150, 50, 130, 30);
       t2 = new JTextField();
       t2.setBounds(450, 100, 130, 30);
       t3 = new JTextField();
       t3.setBounds(450, 150, 130, 30);
       t4 = new JTextField();
       t4.setBounds(100, 150, 130, 30);
       t5 = new JTextField();
       t5.setBounds(450, 200, 130, 30);
       t6 = new JTextField();
       t6.setBounds(100, 200, 130, 30);
       t7 = new JTextField();
       t7.setBounds(450, 250, 130, 30);

       // ComboBoxes
       String[] branchOptions = { " ", "CSE", "ECE", "EEE", "CIVIL", "MECH", "Others" };
       String[] divisionOptions = { " ", "Div-A", "Div-B", "Div-C" };

       j1 = new JComboBox<>(branchOptions);
       j1.setBounds(100, 100, 100, 30);
       j2 = new JComboBox<>(divisionOptions);
       j2.setBounds(470, 50, 140, 30);

       // Buttons
       b1 = new JButton("Save");
       b1.setBounds(150, 300, 70, 30);
       b2 = new JButton("Close");
       b2.setBounds(420, 300, 70, 30);

       // Save Button Action
       b1.addActionListener(new ActionListener() {
           public void actionPerformed(ActionEvent e) {
               String s1 = t1.getText();
               String s2 = t2.getText();
               String s3 = (String) j1.getSelectedItem();
               String s4 = (String) j2.getSelectedItem();
               String s5 = t3.getText();
               String s6 = t4.getText();
               String s7 = t5.getText();
               String s8 = t6.getText();
               String s9 = t7.getText();

               String res = "";
               res += "Name: " + s1 + " | ";
               res += "Roll no: " + s2 + " | ";
               res += "Branch: " + s3 + " | ";
               res += "Division: " + s4 + " | ";
               res += "Marks MC: " + s5 + " | ";
               res += "Marks EMF: " + s6 + " | ";
               res += "Marks DC: " + s7 + " | ";
               res += "Marks FJP: " + s8 + " | ";
               res += "Marks DBMS: " + s9 + " | ";

               JOptionPane.showMessageDialog(f, "Successfully Saved :)");
               f.dispose();
               new ResultForm(res);
           }
       });

       // Close Button Action
       b2.addActionListener(new ActionListener() {
           public void actionPerformed(ActionEvent e) {
               f.dispose();
           }
       });

       // Window Closing
       f.addWindowListener(new WindowAdapter() {
           public void windowClosing(WindowEvent e) {
               System.exit(0);
           }
       });

       // Adding components
       f.add(l1); f.add(t1);
       f.add(l2); f.add(j1);
       f.add(l3); f.add(j2);
       f.add(l4); f.add(t2);
       f.add(l5); f.add(t3);
       f.add(l6); f.add(t4);
       f.add(l7); f.add(t5);
       f.add(l8); f.add(t6);
       f.add(l9); f.add(t7);
       f.add(b1); f.add(b2);

       f.setLayout(null);
       f.setSize(700, 400);
       f.setVisible(true);
   }

   public static void main(String[] args) {
       StudentInfo();
   }
}

class ResultForm extends JFrame {
   public ResultForm(String res) {
       super("Result Form");
       Label l1 = new Label(res);
       add(l1);
       setLayout(new FlowLayout());
       setSize(700, 200);
       setVisible(true);
   }
}
